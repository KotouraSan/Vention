    DEBIAN/
        control      #metadata
    opt/
        hello-anyops.txt     #transfer data



dpkg-deb --build hello-anyops    # build .deb package

mkdir /opt/anyops-repo         # copy .deb to this dir

dpkg-scanpackages . /dev/null > Packages       # scan all .deb packages and build final "Packages" file

gzip -kf Packages       # shrink into .gz arcive


#setup nginx
server {
    listen 80;

    root /opt/anyops-repo;          # dir with all files, Packages, .deb etc.

    location / {
        autoindex on;
    }
}
